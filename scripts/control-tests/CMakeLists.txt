cmake_minimum_required(VERSION 3.20)
project(FubarControlTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
if(NOT FUBAR_BUILD)
  get_filename_component(FUBAR_BUILD "${CMAKE_CURRENT_SOURCE_DIR}/../../build-fubar-msvc" ABSOLUTE)
endif()
file(TO_CMAKE_PATH "${FUBAR_BUILD}" FUBAR_BUILD)
add_executable(control_admission test_control_admission.cpp)
target_compile_definitions(control_admission PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(control_admission PRIVATE /utf-8 /W4)
set_property(TARGET control_admission PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
foreach(source web_server sdr_town_bridge sdr_public_boundary live_hub live_slot live_mp3 fubar_net)
  target_sources(control_admission PRIVATE "${FUBAR_BUILD}/fubar.dir/Release/${source}.obj")
endforeach()
target_link_libraries(control_admission PRIVATE "${FUBAR_BUILD}/Release/shine.lib" ole32 uuid winmm comctl32 shell32 ws2_32 winhttp)
enable_testing()
add_test(NAME control_admission COMMAND control_admission)
add_library(legacy_bridge_fixture SHARED legacy_bridge_fixture.cpp)
set_property(TARGET legacy_bridge_fixture PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
add_executable(bridge_response test_bridge_response.cpp
  "${FUBAR_BUILD}/fubar.dir/Release/sdr_town_bridge.obj"
  "${FUBAR_BUILD}/fubar.dir/Release/fubar_net.obj")
target_compile_definitions(bridge_response PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
set_property(TARGET bridge_response PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
target_link_libraries(bridge_response PRIVATE ws2_32 winhttp)
add_test(NAME bridge_response COMMAND bridge_response "$<TARGET_FILE:legacy_bridge_fixture>")
